# The neutral card layout catalogue is embedded for native and WebAssembly use.
set(CARD_LAYOUTS_JSON ${PROJECT_SOURCE_DIR}/formats/oos-layouts.json)
set(CARD_LAYOUTS_SOURCE ${CMAKE_CURRENT_BINARY_DIR}/generated/CardLayoutsText.cpp)
add_custom_command(
    OUTPUT ${CARD_LAYOUTS_SOURCE}
    COMMAND ${CMAKE_COMMAND} -DINPUT=${CARD_LAYOUTS_JSON} -DOUTPUT=${CARD_LAYOUTS_SOURCE}
            -DNAMESPACE=oor::model -DFUNCTION=cardLayoutsText -P ${PROJECT_SOURCE_DIR}/cmake/EmbedText.cmake
    DEPENDS ${CARD_LAYOUTS_JSON} ${PROJECT_SOURCE_DIR}/cmake/EmbedText.cmake
    COMMENT "Embedding the card layout catalogue"
    VERBATIM)
